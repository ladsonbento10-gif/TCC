CREATE TABLE continentes (
    id_continente INT AUTO_INCREMENT PRIMARY KEY,
    nome_continente VARCHAR(200) NOT NULL UNIQUE,
    codigo_iso CHAR(2) UNIQUE,
    area_km2 DECIMAL(15,2) NOT NULL,
    populacao_estimada BIGINT NOT NULL,
    slug VARCHAR(50) NOT NULL UNIQUE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE paises (
    id_paises INT AUTO_INCREMENT PRIMARY KEY,
    nome_paises VARCHAR(100) NOT NULL,
    sigla_iso2 CHAR(2) NOT NULL UNIQUE,
    sigla_iso3 CHAR(3) NOT NULL UNIQUE,
    codigo_tel VARCHAR(5) NOT NULL,
    continente_id INT NOT NULL,
    CONSTRAINT fk_pais FOREIGN KEY (continente_id) REFERENCES continentes(id_continente)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE estados (
    id_estado INT AUTO_INCREMENT PRIMARY KEY,
    nome_estados VARCHAR(100) NOT NULL,
    uf CHAR(2) NOT NULL,
    pais_id INT NOT NULL,
    CONSTRAINT fk_estado FOREIGN KEY (pais_id) REFERENCES paises(id_paises)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE cidades (
    id_cidade INT AUTO_INCREMENT PRIMARY KEY,
    nome_cidades VARCHAR(100) NOT NULL,
    ibge_code INT NOT NULL,
    estado_id INT NOT NULL,
    CONSTRAINT fk_cidade FOREIGN KEY (estado_id) REFERENCES estados(id_estado)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

CREATE TABLE bairros (
    id_bairro INT AUTO_INCREMENT PRIMARY KEY,
    nome_bairro VARCHAR(100) NOT NULL,
    cidade_id INT NOT NULL,
